POST /chat/stream Stream agent responses (SSE)
Request Body
{ "thread_id": "string", "message": "string" }
SSE Events
data: {"event": "llm", "content": "Hello!", "tool_calls": []}
data: {"event": "llm", "content": "", "tool_calls": [{"id": "...", "name": "query_databricks_sql", "args": {...}}]}
data: {"event": "tools", "tool_call_id": "...", "name": "query_databricks_sql", "output": "col1 | col2\nval1 | val2"}
data: {"event": "llm", "content": "Here are the results...", "tool_calls": []}
data: {"event": "done", "content": "Here are the results..."}
Example
curl
JavaScript
curl -N -X POST /chat/stream \
  -H "Content-Type: application/json" \
  -d '{"thread_id": "my-thread", "message": "Hello!"}'
const resp = await fetch("/chat/stream", {
  method: "POST",
  headers: { "Content-Type": "application/json" },
  body: JSON.stringify({
    thread_id: "my-thread",
    message: "What tables are available?"
  })
});

const reader = resp.body.getReader();
const decoder = new TextDecoder();

while (true) {
  const { done, value } = await reader.read();
  if (done) break;
  const lines = decoder.decode(value).split("\n");
  for (const line of lines) {
    if (line.startsWith("data: ")) {
      const event = JSON.parse(line.slice(6));
      console.log(event.event, event);
    }
  }
}
